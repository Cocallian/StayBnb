Quick Razor/MVC vocabulary used below
  - Controller : C# class that receives a request (URL) and decides which view to show.
  - Action     : a public method on a controller; one action = one URL.
  - View       : a .cshtml file. Razor lets you mix HTML with C# using the @ symbol.
  - Model      : the C# object handed to a view (@model declares its type).
  - asp-controller / asp-action / asp-route-id : "tag helpers" that build the right URL
    for a link or form so you never hard-code paths.

using System.Diagnostics;                 Gives access to Activity (used by the Error action).
using Microsoft.AspNetCore.Mvc;           Controller, IActionResult, View(), etc.
using Microsoft.EntityFrameworkCore;      Include() and ToListAsync() for database queries.
using StayBnb.Data;                       Brings in ApplicationDbContext (the database).
using StayBnb.Models;                     Brings in HostProperty, ErrorViewModel, etc.

namespace StayBnb.Controllers             Groups the class under the project's controllers namespace.
{
    public class HomeController : Controller    Handles the "/" and "/Home/..." URLs.
    {
        private readonly ILogger<HomeController> _logger;     Logger (was already there).
        private readonly ApplicationDbContext _context;       The database connection. Previously
                                                              declared but never set, which was a bug.

        public HomeController(ILogger<HomeController> logger, ApplicationDbContext context)
        {                                  ASP.NET "dependency injection" supplies both arguments
            _logger = logger;              automatically. Added "context" so we can query the DB.
            _context = context;
        }

        public async Task<IActionResult> Index(string? city, decimal? maxPrice, string? q)
        {   The three parameters are filled automatically from the query string, e.g.
            /?city=Cape+Town&maxPrice=3000&q=beach. They are null when not in the URL.

            var query = _context.HostProperties.Where(p => p.IsActive);
                Starts a query on the HostProperties table, only active listings. Nothing is
                sent to SQL Server yet - it is just being built up.

            if (!string.IsNullOrWhiteSpace(city))
                query = query.Where(p => p.City == city);
                If a city was chosen, add "WHERE City = @city" to the SQL.

            if (maxPrice.HasValue)
                query = query.Where(p => p.PricePerNight <= maxPrice.Value);
                If a price was chosen, add "AND PricePerNight <= @maxPrice".

            if (!string.IsNullOrWhiteSpace(q))
                query = query.Where(p => p.Title.Contains(q) || p.Description.Contains(q) || p.Address.Contains(q));
                If search text was typed, add a LIKE '%text%' test on title, description, address.

            var properties = await query
                .Include(p => p.Images)      Also load each property's images (JOIN).
                .Include(p => p.Reviews)     Also load each property's reviews (for the rating).
                .ToListAsync();              NOW the SQL runs; "await" frees the thread while waiting.
            return View(properties);         Passes the list to Views/Home/Index.cshtml as its Model.
        }

        Privacy() and Error() are the original template actions and were not changed.
    }
}

5. Controllers/HostApplicationController.cs

using System.Security.Claims;                 ClaimTypes.NameIdentifier = the logged-in user's id.
using Microsoft.AspNetCore.Authorization;     The [Authorize] attribute.

public class HostApplicationController : Controller
    constructor stores the database, same pattern as above.

    // Status: used by the "Host Application Status" nav link.
    public async Task<IActionResult> Status()
        var userId = User.FindFirstValue(ClaimTypes.NameIdentifier);
            "User" is the current visitor. This returns their id, or null if not logged in.
        if (userId == null) return RedirectToAction("Login", "Account");
            Not logged in -> go to the login page.
        var application = await _context.HostApplications
            .Where(a => a.ApplicationUserId == userId)       Only this user's applications,
            .OrderByDescending(a => a.AppliedAt)             newest first,
            .FirstOrDefaultAsync();                          take the newest (or null).
        if (application == null) return RedirectToAction(nameof(Apply));
            Never applied -> go to the Apply page. nameof(Apply) = the text "Apply" (typo-proof).
        return View(application);        Has applied -> show the Status view with that application.

    [Authorize]                           Visitors who are not logged in are sent to /Account/Login
                                          automatically (LoginPath is set in Program.cs).
    public async Task<IActionResult> Apply()
        var userId = ...;                 Same user id lookup.
        if (await _context.HostApplications.AnyAsync(a => a.ApplicationUserId == userId))
            return RedirectToAction(nameof(Status));    Already applied -> show status instead.
        return View();                    Otherwise show the Apply page.

8. Controllers/AccountController.cs
=======================================================================
private readonly UserManager<ApplicationUser> _userManager;
    ASP.NET Identity's tool for creating users, finding users, and reading their roles.
private readonly SignInManager<ApplicationUser> _signInManager;
    Identity's tool for checking a password and creating/removing the login cookie.
Constructor receives both via dependency injection and stores them.

[HttpGet] Login(string? returnUrl = null)
    Runs when the page is opened. Returns the Login view with a model that remembers returnUrl.

[HttpPost, ValidateAntiForgeryToken] Login(LoginViewModel model)
    Runs when the form is submitted. [HttpPost] limits it to POST requests;
    ValidateAntiForgeryToken rejects forged cross-site posts (the form tag helper adds the token).
    if (!ModelState.IsValid) return View(model);
        Validation attributes failed (empty email, bad format...) -> redisplay the form with errors.
    var result = await _signInManager.PasswordSignInAsync(model.Email, model.Password, isPersistent: false, lockoutOnFailure: false);
        Checks the password and, if correct, sets the login cookie. The user name is the email.
        isPersistent:false = cookie ends when the browser closes. lockoutOnFailure:false = no lockout.
    if (!result.Succeeded) { ModelState.AddModelError(string.Empty, "Invalid email or password."); return View(model); }
        Wrong login -> show a generic message (does not reveal which part was wrong).
    var user = await _userManager.FindByEmailAsync(model.Email);
    return await RedirectByRoleAsync(user!, model.ReturnUrl);
        Loads the user and sends them to the right page for their role (see helper below).

[HttpGet] Register()      Shows the empty signup form.

[HttpPost] Register(RegisterViewModel model)
    if (!ModelState.IsValid) return View(model);       Bad input -> redisplay with errors.
    var user = new ApplicationUser { UserName = model.Email, Email = ..., FirstName = ..., LastName = ... };
        Builds the user object. Identity uses UserName to log in; we use the email for it.
    var result = await _userManager.CreateAsync(user, model.Password);
        Saves the user; Identity hashes the password (never stored as plain text).
    if (!result.Succeeded) { add every error to ModelState; return View(model); }
        Shows problems such as "email already taken" or password rule failures.
    await _userManager.AddToRoleAsync(user, Roles.Guest);
        Every new account starts as a Guest.
    await _signInManager.SignInAsync(user, isPersistent: false);   Logs them in immediately.
    return RedirectToAction("Index", "Home");                      Sends them to the home page.

[HttpPost] Logout()
    await _signInManager.SignOutAsync();     Deletes the login cookie.
    return RedirectToAction("Index", "Home");

AccessDenied()   Shown when a logged-in user opens a page their role is not allowed to see
                 (path configured in Program.cs).

private async Task<IActionResult> RedirectByRoleAsync(ApplicationUser user, string? returnUrl)
    if (!string.IsNullOrEmpty(returnUrl) && Url.IsLocalUrl(returnUrl)) return Redirect(returnUrl);
        If the user was sent to login from a specific page, go back there. IsLocalUrl blocks
        redirects to other websites (open-redirect attack protection).
    if user is SuperAdmin or Admin -> Admin / Dashboard
    if user is Host                -> Host / Dashboard
    otherwise (Guest)              -> Home / Index

9. AdminController, HostController and dashboards
=======================================================================
[Authorize(Roles = Roles.Admin + "," + Roles.SuperAdmin)] on AdminController
    Only users in either of those roles may reach any action in the controller. Others get the
    login page (if logged out) or the Access Denied page (if logged in with the wrong role).
[Authorize(Roles = Roles.Host)] on HostController     Same idea for hosts.
Dashboard() returns View()   Shows Views/Admin/Dashboard.cshtml or Views/Host/Dashboard.cshtml,
                             which are just a title and an <h2> heading for now.
Roles.Admin etc. are the string constants in Models/Roles.cs ("Admin", "Host", ...).

    var isAdmin = roles.Contains(Roles.Admin) || roles.Contains(Roles.SuperAdmin);

          Simple true/false flags used to choose which links to show.
    var cities = await Db.HostProperties.Where(p => p.IsActive).Select(p => p.City).Distinct().OrderBy(c => c).ToListAsync();
          Every distinct city that has an active listing, alphabetical -> fills the city dropdown.
